__global__
void detectPlants(float *A, int *result, int M, int N)
{
    int row = blockIdx.x * blockDim.x + threadIdx.x;

    if (row >= M)
        return;

    int lastWater = -1;

    for(int j = 0; j < N; j++)
    {
        if(j == 0)
        {
            if(A[row*N + j] >= 0.5)
                lastWater = 0;
        }
        else
        {
            if(A[row*N + j] >= 0.5 ||
               A[row*N + j] > A[row*N + j - 1])
            {
                lastWater = j;
            }
        }
    }

    if(lastWater == -1)
        result[row] = N;
    else
        result[row] = N - 1 - lastWater;
}

__global__
void checkPassword(const char *db,
                   const char *password,
                   int n,
                   int *found)
{
    int id = blockIdx.x * blockDim.x + threadIdx.x;

    if (id >= n)
        return;

    bool equal = true;

    for(int i=0;i<32;i++)
    {
        if(db[id*32+i] != password[i])
        {
            equal = false;
            break;
        }
    }

    if(equal)
        *found = 1;
}
